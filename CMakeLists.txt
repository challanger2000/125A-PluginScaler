cmake_minimum_required(VERSION 3.24)
project(PluginScaler VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(core-tests tests/core.cpp)
target_include_directories(core-tests PRIVATE include)
add_test(NAME core-contracts COMMAND core-tests)
add_test(NAME release-assertion-negative COMMAND core-tests intentional-failure)
set_tests_properties(release-assertion-negative PROPERTIES WILL_FAIL TRUE)
if(WIN32)
  set(SCALER_VST3_ROOT "" CACHE PATH "Directory containing official pluginterfaces checkout")
  if(NOT SCALER_VST3_ROOT)
    include(FetchContent)
    FetchContent_Declare(pluginterfaces
      GIT_REPOSITORY https://github.com/steinbergmedia/vst3_pluginterfaces.git
      GIT_TAG 4f547e8e102b47de4a8b8aaf343c73b700786372
      SOURCE_DIR "${CMAKE_BINARY_DIR}/_deps/pluginterfaces"
      SOURCE_SUBDIR unused)
    FetchContent_MakeAvailable(pluginterfaces)
    set(SCALER_VST3_ROOT "${CMAKE_BINARY_DIR}/_deps")
  endif()
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MINGW)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
      add_compile_definitions(_M_AMD64=100)
    else()
      add_compile_definitions(_M_IX86=600)
    endif()
    add_link_options(-static -static-libgcc -static-libstdc++)
    target_link_options(core-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_library(scaler-bridge STATIC src/bridge.cpp)
  target_include_directories(scaler-bridge PUBLIC include "${SCALER_VST3_ROOT}")
  add_executable(scaler-helper src/helper.cpp src/vst2_adapter.cpp src/vst3_adapter.cpp)
  target_include_directories(scaler-helper PRIVATE include "${SCALER_VST3_ROOT}")
  target_link_libraries(scaler-helper PRIVATE user32 gdi32 shell32)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(scaler-helper PROPERTIES OUTPUT_NAME scaler-helper64)
  else()
    set_target_properties(scaler-helper PROPERTIES OUTPUT_NAME scaler-helper32)
  endif()
  add_library(scaler-fixture SHARED tests/fixture.cpp)
  target_include_directories(scaler-fixture PRIVATE include)
  target_link_libraries(scaler-fixture PRIVATE user32 gdi32)
  set_target_properties(scaler-fixture PROPERTIES PREFIX "")
  if(MINGW)
    target_link_options(scaler-helper PRIVATE -municode)
  endif()
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_library(PluginScaler MODULE src/plugin.cpp src/editor.cpp)
    target_link_libraries(PluginScaler PRIVATE scaler-bridge user32 gdi32)
    set_target_properties(PluginScaler PROPERTIES PREFIX "" SUFFIX ".vst3")
    add_executable(integration-tests tests/integration.cpp)
    target_link_libraries(integration-tests PRIVATE scaler-bridge user32 gdi32)
    add_library(fixture-vst3 MODULE tests/fixture_vst3.cpp)
    target_include_directories(fixture-vst3 PRIVATE include "${SCALER_VST3_ROOT}")
    set_target_properties(fixture-vst3 PROPERTIES PREFIX "" SUFFIX ".vst3")
    add_executable(adapter-tests tests/adapter.cpp)
    target_link_libraries(adapter-tests PRIVATE scaler-bridge)
    if(MINGW)
      target_link_options(integration-tests PRIVATE -municode)
      target_link_options(adapter-tests PRIVATE -municode)
    endif()
    add_test(NAME packaged-vst3-host COMMAND integration-tests "$<TARGET_FILE:PluginScaler>")
    add_test(NAME hosted-vst3-adapter COMMAND adapter-tests "$<TARGET_FILE:fixture-vst3>")
    set_tests_properties(hosted-vst3-adapter PROPERTIES TIMEOUT 30)
    set_tests_properties(packaged-vst3-host PROPERTIES TIMEOUT 90 WORKING_DIRECTORY "$<TARGET_FILE_DIR:PluginScaler>")
  endif()
endif()
